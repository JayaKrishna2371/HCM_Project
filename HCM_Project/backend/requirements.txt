from apscheduler.schedulers.background import BackgroundScheduler
from app.db.database import SessionLocal
from app.repositories.clouds.vmware.vmware_repository import VmwareRepository
from app.services.clouds.vmware.vmware_service import VmwareService
import logging

logger = logging.getLogger(__name__)

scheduler = BackgroundScheduler()

def run_auto_discovery():
"""Runs discovery for all endpoints every 5 minutes"""
db = SessionLocal()
try:
endpoints = VmwareRepository.get_all_endpoints(db)

for endpoint in endpoints:
try:
logger.info(f"Auto-discovery started for endpoint {endpoint.id}")
VmwareService.discover_inventory(endpoint, db)
logger.info(f"Auto-discovery completed for endpoint {endpoint.id}")
except Exception as e:
logger.error(f"Auto-discovery failed for endpoint {endpoint.id}: {str(e)}")
except Exception as e:
logger.error(f"Auto-discovery scheduler error: {str(e)}")
finally:
db.close()

def start_scheduler():
"""Start the background discovery scheduler"""
if not scheduler.running:
scheduler.add_job(
run_auto_discovery,
'interval',
minutes=5,
id='vmware_auto_discovery',
name='VMware Auto-Discovery',
misfire_grace_time=60
)
scheduler.start()
logger.info("VMware auto-discovery scheduler started (every 5 minutes)")

def stop_scheduler():
"""Stop the scheduler gracefully"""
if scheduler.running:
scheduler.shutdown()
logger.info("VMware auto-discovery scheduler stopped")


@staticmethod
def get_all_endpoints(db: Session):
"""Get all VMware endpoints"""
from app.models.clouds.vmware.vmware_model import VmwareEndpoint
return db.query(VmwareEndpoint).all()


from contextlib import asynccontextmanager
from app.services.clouds.vmware.discovery_scheduler import start_scheduler, stop_scheduler


@asynccontextmanager
async def lifespan(app: FastAPI):
# Startup
start_scheduler()
yield
# Shutdown
stop_scheduler()

app = FastAPI(
title="HCM API",
lifespan=lifespan
)


auto_discovery_enabled = Column(Boolean, default=True)
