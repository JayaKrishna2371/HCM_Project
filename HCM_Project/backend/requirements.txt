# vmware_service.py
@staticmethod
def get_templates(endpoint_id, db):
service_instance = VmwareService._get_instance(endpoint_id, db) # match get_vms' helper name
return VCenterClient().get_templates(service_instance)

# vmware_controller.py (static method + route)
@staticmethod
def get_templates(endpoint_id, db):
return VmwareService.get_templates(endpoint_id, db)

@router.get("/endpoints/{endpoint_id}/templates")
def get_templates(endpoint_id: int, db: Session = Depends(get_db)):
return VmwareController.get_templates(endpoint_id, db)
