@staticmethod
def save_templates(endpoint_id: int, templates: list, db: Session):
for t in templates:
db.add(
VmwareTemplate(
endpoint_id=endpoint_id,
name=t.get("name"),
guest_os=t.get("guest_os"),
cpu_count=t.get("cpu"),
memory_mb=t.get("memory_mb"),
disk_gb=t.get("disk_gb"),
)
)
db.commit()

@staticmethod
def clear_templates(endpoint_id: int, db: Session):
db.query(VmwareTemplate).filter(
VmwareTemplate.endpoint_id == endpoint_id
).delete()
db.commit()

@staticmethod
def get_templates(endpoint_id: int, db: Session):
return (
db.query(VmwareTemplate)
.filter(VmwareTemplate.endpoint_id == endpoint_id)
.all()
)
