import requests
from app.core.config import settings

BASE = f"{settings.VMWARE_SERVICE_URL}/api/v1/vmware"


def _headers(auth_token: str) -> dict:
return {"Authorization": f"Bearer {auth_token}"}


def power_on_vm(endpoint_id: int, vm_name: str, auth_token: str) -> dict:
resp = requests.post(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}/power-on",
headers=_headers(auth_token)
)
return resp.json()


def power_off_vm(endpoint_id: int, vm_name: str, auth_token: str) -> dict:
resp = requests.post(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}/power-off",
headers=_headers(auth_token)
)
return resp.json()


def reboot_vm(endpoint_id: int, vm_name: str, auth_token: str) -> dict:
resp = requests.post(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}/reboot",
headers=_headers(auth_token)
)
return resp.json()


def delete_vm(endpoint_id: int, vm_name: str, auth_token: str) -> dict:
resp = requests.delete(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}",
headers=_headers(auth_token)
)
return resp.json()


def reconfigure_vm_resources(endpoint_id: int, vm_name: str, auth_token: str,
cpu_count: int = None, memory_mb: int = None) -> dict:
payload = {}
if cpu_count is not None:
payload["cpu_count"] = cpu_count
if memory_mb is not None:
payload["memory_mb"] = memory_mb
resp = requests.post(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}/resources",
json=payload,
headers=_headers(auth_token)
)
return resp.json()


def add_disk(endpoint_id: int, vm_name: str, auth_token: str, size_gb: int) -> dict:
resp = requests.post(
f"{BASE}/endpoints/{endpoint_id}/vms/{vm_name}/disks",
json={"size_gb": size_gb},
headers=_headers(auth_token)
)
return resp.json()


def list_vms(endpoint_id: int, auth_token: str) -> list:
resp = requests.get(
f"{BASE}/endpoints/{endpoint_id}/vms",
headers=_headers(auth_token)
)
return resp.json()


def list_templates(endpoint_id: int, auth_token: str) -> list:
resp = requests.get(
f"{BASE}/endpoints/{endpoint_id}/templates",
headers=_headers(auth_token)
)
return resp.json()

import ollama
from app.core.config import settings
from app.mcp_tools.tool_definitions import TOOLS
from app.mcp_tools.vmware_tools import TOOL_FUNCTIONS

DESTRUCTIVE_TOOLS = {"delete_vm", "power_off_vm"}

client = ollama.Client(host=settings.OLLAMA_URL)


def handle_chat(user_message: str, history: list, auth_token: str) -> str:
messages = [{"role": h.role, "content": h.content} for h in history]
messages.append({"role": "user", "content": user_message})

response = client.chat(
model=settings.OLLAMA_MODEL,
messages=messages,
tools=TOOLS
)

assistant_message = response["message"]
tool_calls = assistant_message.get("tool_calls")

if not tool_calls:
return assistant_message["content"]

messages.append(assistant_message)

for call in tool_calls:
tool_name = call["function"]["name"]
args = call["function"]["arguments"]

if tool_name in DESTRUCTIVE_TOOLS:
vm_name = args.get("vm_name", "this VM")
return (f"⚠️ You asked me to run '{tool_name}' on '{vm_name}'. "
f"This is a destructive action. Please confirm by replying "
f"'yes, {tool_name} {vm_name}' to proceed.")

func = TOOL_FUNCTIONS.get(tool_name)
if not func:
result = {"error": f"Unknown tool: {tool_name}"}
else:
try:
result = func(**args, auth_token=auth_token)
except Exception as e:
result = {"error": str(e)}

messages.append({"role": "tool", "content": str(result)})

final_response = client.chat(
model=settings.OLLAMA_MODEL,
messages=messages,
tools=TOOLS
)

return final_response["message"]["content"]



from fastapi import APIRouter, Header, HTTPException
from app.schemas.chat_schema import ChatRequest, ChatResponse
from app.services.chat_service import handle_chat

router = APIRouter(prefix="/api/v1/chat", tags=["AI Chat"])


@router.post("/", response_model=ChatResponse)
async def chat(req: ChatRequest, authorization: str = Header(None)):
if not authorization or not authorization.startswith("Bearer "):
raise HTTPException(status_code=401, detail="Missing or invalid Authorization header")

auth_token = authorization.replace("Bearer ", "")
reply = handle_chat(req.message, req.history, auth_token=auth_token)
return ChatResponse(reply=reply)
