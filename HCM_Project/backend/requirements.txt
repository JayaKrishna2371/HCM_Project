loadTemplates() {
(this.vmwareService.getTemplates(this.endpointId) as any)
.subscribe((res: any) => {
this.templates = res || [];
this.templateCount = this.templates.length;
this.cdr.detectChanges();
});
}

templateCount: number = 0;
