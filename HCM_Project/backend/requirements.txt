def set_vm_nic_connected(self, service_instance, vm_name, connected):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"success": False, "message": "VM not found"}

nic = None
for dev in vm.config.hardware.device:
if isinstance(dev, vim.vm.device.VirtualEthernetCard):
nic = dev
break
if not nic:
return {"success": False, "message": "No network adapter found on VM"}

nic_spec = vim.vm.device.VirtualDeviceSpec()
nic_spec.operation = vim.vm.device.VirtualDeviceSpec.Operation.edit
nic_spec.device = nic

nic_spec.device.connectable = vim.vm.device.VirtualDevice.ConnectInfo()
nic_spec.device.connectable.startConnected = connected
nic_spec.device.connectable.connected = connected
nic_spec.device.connectable.allowGuestControl = True

config_spec = vim.vm.ConfigSpec()
config_spec.deviceChange = [nic_spec]

task = vm.ReconfigVM_Task(spec=config_spec)
self.wait_for_task(task)

state = "connected" if connected else "disconnected"
return {"success": True, "message": f"NIC {state}"}

toggleNicConnection() {
if (!this.manageVm) return;
const newState = !this.nicConnected;
const vmName = this.manageVm.name;
this.handle(
this.manageVm,
this.vmwareService.setVmNicConnected(this.endpointId, vmName, newState),
newState ? 'NIC connected' : 'NIC disconnected',
'NIC update failed'
);
this.nicConnected = newState;
}
