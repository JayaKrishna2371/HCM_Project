def convert_template_to_vm(self, service_instance, template_name):
vm = self.find_vm(service_instance, template_name)
if not vm:
return {"success": False, "message": "Template not found"}
if not vm.config.template:
return {"success": False, "message": "Not a template"}

# find the resource pool from the template's host/cluster
host = vm.runtime.host
pool = host.parent.resourcePool # cluster's root pool
task = vm.MarkAsVirtualMachine(pool=pool, host=host)
self.wait_for_task(task)
return {"success": True, "message": f"{template_name} converted to VM"}


def convert_vm_to_template(self, service_instance, vm_name):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"success": False, "message": "VM not found"}
if vm.config.template:
return {"success": False, "message": "Already a template"}
if vm.runtime.powerState != vim.VirtualMachinePowerState.poweredOff:
return {"success": False, "message": "Power off the VM before converting to template"}

task = vm.MarkAsTemplate()
self.wait_for_task(task)
return {"success": True, "message": f"{vm_name} converted to template"}



@staticmethod
def convert_template_to_vm(endpoint_id, template_name, db):
endpoint = VmwareRepository.get_endpoint_by_id(endpoint_id, db)
if not endpoint:
raise HTTPException(status_code=404, detail="Endpoint not found")
service_instance = VmwareService._get_service_instance(endpoint)
return VCenterClient().convert_template_to_vm(service_instance, template_name)

this.handle(
template,

'Converted to VM', 'Convert to VM failed'
);
}

convertToTemplate() {
if (!this.manageVm) return;
this.handle(
this.manageVm,
this.vmwareService.convertVmToTemplate(this.endpointId, this.manageVm.name),
'Converted to template', 'Convert to template failed'
);
this.closeManage();
}

isDiscovering = false;

triggerDiscovery() {
this.isDiscovering = true;
this.vmwareService.discover(this.endpointId).subscribe(
(res) => {
this.showToast('Discovery in progress...', 'info');
setTimeout(() => {
this.loadInventory();
this.loadTemplates();
this.isDiscovering = false;
this.showToast('Discovery complete', 'success');
}, 2000);
},
(err) => {
this.isDiscovering = false;
this.showToast('Discovery failed', 'error');
}
);
}
