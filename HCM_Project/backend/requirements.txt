vmNics: any[] = []; // all NICs on the VM
availableNetworks: any[] = []; // host portgroups for the dropdown
selectedNicKey: number | null = null;
selectedNetwork: string | null = null;
currentNetwork: string | null = null;
nicConnected: boolean = true;


(this.vmwareService.getVmNetworks(this.endpointId, vm.name) as any)
.subscribe((res: any) => {
this.vmNics = res.nics || [];
this.availableNetworks = res.networks || [];
if (this.vmNics.length) {
this.onNicSelect(this.vmNics[0].key); // preselect first NIC
}
this.cdr.detectChanges();
});


onNicSelect(nicKey: number) {
this.selectedNicKey = nicKey;
const nic = this.vmNics.find(n => n.key === nicKey);
if (nic) {
this.currentNetwork = nic.current_network;
this.selectedNetwork = nic.current_network;
this.nicConnected = nic.connected;
}
}


applyNetwork() {
if (!this.manageVm || this.selectedNicKey === null) return;
if (!this.selectedNetwork) { this.showToast('Select a network', 'error'); return; }
if (this.selectedNetwork === this.currentNetwork) { this.showToast('No change', 'error'); return; }
const vmName = this.manageVm.name;
this.handle(
this.manageVm,
this.vmwareService.changeVmNetwork(this.endpointId, vmName, this.selectedNetwork, this.selectedNicKey),
'Network changed', 'Network change failed'
);
}

toggleNicConnection() {
if (!this.manageVm || this.selectedNicKey === null) return;
const newState = !this.nicConnected;
const vmName = this.manageVm.name;
this.handle(
this.manageVm,
this.vmwareService.setVmNicConnected(this.endpointId, vmName, newState, this.selectedNicKey),
newState ? 'NIC connected' : 'NIC disconnected', 'NIC update failed'
);
this.nicConnected = newState;
}
