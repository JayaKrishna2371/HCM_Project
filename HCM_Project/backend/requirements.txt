def get_vms(self, service_instance):
content = service_instance.RetrieveContent()
container = content.viewManager.CreateContainerView(
content.rootFolder, [vim.Datacenter], True
)

vms = []

try:
for datacenter in container.view:
clusters = datacenter.hostFolder.childEntity
for cluster in clusters:
hosts = cluster.host

for host in hosts:
if not hasattr(host, 'vm') or not host.vm:
continue

# Batch-fetch properties for all VMs on this host
for vm in host.vm:
try:
config = vm.config
runtime = vm.runtime

# Skip templates
if not config or config.template:
continue

# Calculate disk
disk_gb = 0
if config.hardware and config.hardware.device:
for device in config.hardware.device:
if hasattr(device, "capacityInKB"):
disk_gb += device.capacityInKB / (1024 * 1024)

vms.append({
"name": vm.name,
"power_state": str(runtime.powerState) if runtime else None,
"cpu_count": config.hardware.numCPU if config.hardware else None,
"memory_mb": config.hardware.memoryMB if config.hardware else None,
"disk_gb": round(disk_gb, 2),
"guest_os": config.guestFullName,
"vm_uuid": config.uuid,
"host_name": host.name,
"cluster_name": cluster.name,
"datacenter_name": datacenter.name,
})
except Exception as e:
continue
except Exception as e:
pass
finally:
container.Destroy()

return vms
