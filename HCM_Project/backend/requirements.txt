def get_vm_host_networks(self, service_instance, vm_name):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"nics": [], "networks": []}

# available portgroups on the host
networks = []
seen = set()
for net in vm.runtime.host.network:
if net.name not in seen:
seen.add(net.name)
net_type = "dvs" if isinstance(net, vim.dvs.DistributedVirtualPortgroup) else "standard"
networks.append({"name": net.name, "type": net_type})

# all NICs on the VM
nics = []
for dev in vm.config.hardware.device:
if isinstance(dev, vim.vm.device.VirtualEthernetCard):
current = None
backing = dev.backing
if isinstance(backing, vim.vm.device.VirtualEthernetCard.DistributedVirtualPortBackingInfo):
pg_key = backing.port.portgroupKey
for net in vm.runtime.host.network:
if isinstance(net, vim.dvs.DistributedVirtualPortgroup) and net.key == pg_key:
current = net.name
break
elif hasattr(backing, "network") and backing.network:
current = backing.network.name
elif hasattr(backing, "deviceName"):
current = backing.deviceName

nics.append({
"key": dev.key,
"label": dev.deviceInfo.label,
"current_network": current,
"connected": dev.connectable.connected if dev.connectable else False,
})

return {"nics": nics, "networks": networks}


def change_vm_network(self, service_instance, vm_name, network_name, nic_key=None):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"success": False, "message": "VM not found"}

target = next((n for n in vm.runtime.host.network if n.name == network_name), None)
if not target:
return {"success": False, "message": "Network not found on host"}

if nic_key is not None:
nic = next((d for d in vm.config.hardware.device
if isinstance(d, vim.vm.device.VirtualEthernetCard) and d.key == nic_key), None)
else:
nic = next((d for d in vm.config.hardware.device
if isinstance(d, vim.vm.device.VirtualEthernetCard)), None)
if not nic:
return {"success": False, "message": "No NIC on VM"}

if isinstance(target, vim.dvs.DistributedVirtualPortgroup):
pc = vim.dvs.PortConnection()
pc.portgroupKey = target.key
pc.switchUuid = target.config.distributedVirtualSwitch.uuid
nic.backing = vim.vm.device.VirtualEthernetCard.DistributedVirtualPortBackingInfo()
nic.backing.port = pc
else:
nic.backing = vim.vm.device.VirtualEthernetCard.NetworkBackingInfo()
nic.backing.network = target
nic.backing.deviceName = network_name

nic.connectable = vim.vm.device.VirtualDevice.ConnectInfo()
nic.connectable.connected = True
nic.connectable.startConnected = True

spec = vim.vm.device.VirtualDeviceSpec()
spec.operation = vim.vm.device.VirtualDeviceSpec.Operation.edit
spec.device = nic

task = vm.ReconfigVM_Task(spec=vim.vm.ConfigSpec(deviceChange=[spec]))
self.wait_for_task(task)
return {"success": True, "message": f"Network changed to {network_name}"}


def set_vm_nic_connected(self, service_instance, vm_name, connected, nic_key=None):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"success": False, "message": "VM not found"}

if nic_key is not None:
nic = next((d for d in vm.config.hardware.device
if isinstance(d, vim.vm.device.VirtualEthernetCard) and d.key == nic_key), None)
else:
nic = next((d for d in vm.config.hardware.device
if isinstance(d, vim.vm.device.VirtualEthernetCard)), None)
if not nic:
return {"success": False, "message": "No NIC on VM"}

nic.connectable = vim.vm.device.VirtualDevice.ConnectInfo()
nic.connectable.connected = connected
nic.connectable.startConnected = connected
nic.connectable.allowGuestControl = True

spec = vim.vm.device.VirtualDeviceSpec()
spec.operation = vim.vm.device.VirtualDeviceSpec.Operation.edit
spec.device = nic

task = vm.ReconfigVM_Task(spec=vim.vm.ConfigSpec(deviceChange=[spec]))
self.wait_for_task(task)

state = "connected" if connected else "disconnected"
return {"success": True, "message": f"NIC {state}"}

vmNics: any[] = [];
availableNetworks: any[] = [];
selectedNicKey: number | null = null;
selectedNetwork: string | null = null;
currentNetwork: string | null = null;
nicConnected: boolean = true;


(this.vmwareService.getVmNetworks(this.endpointId, vm.name) as any)
.subscribe((res: any) => {
this.vmNics = res.nics || [];
this.availableNetworks = res.networks || [];
if (this.vmNics.length) {
this.onNicSelect(this.vmNics[0].key);
}
this.cdr.detectChanges();
});


onNicSelect(nicKey: number) {
this.selectedNicKey = nicKey;
const nic = this.vmNics.find(n => n.key === nicKey);
if (nic) {
this.currentNetwork = nic.current_network;
this.selectedNetwork = nic.current_network;
this.nicConnected = nic.connected;
}
}

applyNetwork() {
if (!this.manageVm || this.selectedNicKey === null) return;
if (!this.selectedNetwork) { this.showToast('Select a network', 'error'); return; }
if (this.selectedNetwork === this.currentNetwork) { this.showToast('No change', 'error'); return; }
const vmName = this.manageVm.name;
this.handle(
this.manageVm,
this.vmwareService.changeVmNetwork(this.endpointId, vmName, this.selectedNetwork, this.selectedNicKey),
'Network changed', 'Network change failed'
);
}

toggleNicConnection() {
if (!this.manageVm || this.selectedNicKey === null) return;
const newState = !this.nicConnected;
const vmName = this.manageVm.name;
this.handle(
this.manageVm,
this.vmwareService.setVmNicConnected(this.endpointId, vmName, newState, this.selectedNicKey),
newState ? 'NIC connected' : 'NIC disconnected', 'NIC update failed'
);
this.nicConnected = newState;
}
