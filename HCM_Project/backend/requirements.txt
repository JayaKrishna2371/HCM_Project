class VmwareTemplate(Base):

__tablename__ = "vmware_templates"

id = Column(
Integer,
primary_key=True
)

endpoint_id = Column(
Integer,
nullable=False
)

name = Column(
String,
nullable=False
)

guest_os = Column(
String,
nullable=True
)

cpu_count = Column(
Integer,
nullable=True
)

memory_mb = Column(
Integer,
nullable=True
)


@staticmethod
def save_templates(
endpoint_id: int,
templates: list,
db: Session
):
for t in templates:
db.add(
VmwareTemplate(
endpoint_id=endpoint_id,
name=t.get("name"),
guest_os=t.get("guest_os"),
cpu_count=t.get("cpu"),
memory_mb=t.get("memory_mb")
)
)
db.commit()

@staticmethod
def clear_templates(endpoint_id: int, db: Session):
db.query(VmwareTemplate).filter(
VmwareTemplate.endpoint_id == endpoint_id
).delete()
db.commit()

@staticmethod
def get_templates(endpoint_id: int, db: Session):
return (
db.query(VmwareTemplate)
.filter(VmwareTemplate.endpoint_id == endpoint_id)
.all()
)

def get_templates(self, service_instance):
content = service_instance.RetrieveContent()
container = content.viewManager.CreateContainerView(
content.rootFolder, [vim.VirtualMachine], True
)
templates = []
for vm in container.view:
if vm.config and vm.config.template:
disk_gb = 0
try:
for device in vm.config.hardware.device:
if hasattr(device, "capacityInKB"):
disk_gb += device.capacityInKB / (1024 * 1024)
except Exception:
disk_gb = 0

templates.append({
"name": vm.name,
"guest_os": vm.config.guestFullName,
"cpu": vm.config.hardware.numCPU,
"memory_mb": vm.config.hardware.memoryMB,
"disk_gb": round(disk_gb, 2),
})
container.Destroy()
return templates
