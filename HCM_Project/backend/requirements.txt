def get_vm_host_networks(self, service_instance, vm_name):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"current_network": None, "networks": []}

# current portgroup on the first NIC
current = None
for d in vm.config.hardware.device:
if isinstance(d, vim.vm.device.VirtualEthernetCard):
backing = d.backing
if hasattr(backing, "port") and backing.port:
# DVS backing -> resolve portgroup key to name
for net in vm.runtime.host.network:
if getattr(net, "key", None) == backing.port.portgroupKey:
current = net.name
break
elif hasattr(backing, "deviceName"):
current = backing.deviceName
break

networks = []
seen = set()
for net in vm.runtime.host.network:
if net.name not in seen:
seen.add(net.name)
is_dvs = isinstance(net, vim.dvs.DistributedVirtualPortgroup)
networks.append({"name": net.name, "type": "dvs" if is_dvs else "standard"})

return {"current_network": current, "networks": networks}


def change_vm_network(self, service_instance, vm_name, network_name):
vm = self.find_vm(service_instance, vm_name)
if not vm:
return {"success": False, "message": "VM not found"}

target = next((n for n in vm.runtime.host.network if n.name == network_name), None)
if not target:
return {"success": False, "message": "Network not found on host"}

nic = next((d for d in vm.config.hardware.device
if isinstance(d, vim.vm.device.VirtualEthernetCard)), None)
if not nic:
return {"success": False, "message": "No NIC on VM"}

if isinstance(target, vim.dvs.DistributedVirtualPortgroup):
pc = vim.dvs.PortConnection()
pc.portgroupKey = target.key
pc.switchUuid = target.config.distributedVirtualSwitch.uuid
nic.backing = vim.vm.device.VirtualEthernetCard.DistributedVirtualPortBackingInfo()
nic.backing.port = pc
else:
nic.backing = vim.vm.device.VirtualEthernetCard.NetworkBackingInfo()
nic.backing.network = target
nic.backing.deviceName = network_name

nic.connectable = vim.vm.device.VirtualDevice.ConnectInfo()
nic.connectable.connected = True
nic.connectable.startConnected = True

spec = vim.vm.device.VirtualDeviceConfigSpec()
spec.operation = vim.vm.device.VirtualDeviceConfigSpec.Operation.edit
spec.device = nic

task = vm.ReconfigVM_Task(spec=vim.vm.ConfigSpec(deviceChange=[spec]))
self.wait_for_task(task)
return {"success": True, "message": f"Network changed to {network_name}"}
