def get_vms(self, service_instance):
content = service_instance.RetrieveContent()
container = content.viewManager.CreateContainerView(
content.rootFolder, [vim.VirtualMachine], True
)

# PropertyCollector batch fetch (much faster than per-VM loops)
collector = content.propertyCollector

filter_spec = vim.PropertyFilterSpec()
filter_spec.propSet = [vim.PropertySpec(
all=False,
type=vim.VirtualMachine,
pathSet=['config', 'runtime', 'parent']
)]
filter_spec.objectSet = [vim.ObjectSpec(
obj=container,
skip=False
)]

try:
result = collector.RetrievePropertiesEx(
[filter_spec],
vim.RetrieveOptions()
)
except Exception:
container.Destroy()
return []

vms = []

if result and result.objects:
for obj in result.objects:
try:
vm_props = {prop.name: prop.val for prop in obj.propSet}
config = vm_props.get('config')
runtime = vm_props.get('runtime')

# Skip templates
if not config or config.template:
continue

# Calculate disk
disk_gb = 0
if config.hardware and config.hardware.device:
for device in config.hardware.device:
if hasattr(device, "capacityInKB"):
disk_gb += device.capacityInKB / (1024 * 1024)

vms.append({
"name": obj.obj.name,
"power_state": str(runtime.powerState) if runtime else None,
"cpu_count": config.hardware.numCPU if config.hardware else None,
"memory_mb": config.hardware.memoryMB if config.hardware else None,
"disk_gb": round(disk_gb, 2),
"guest_os": config.guestFullName,
"vm_uuid": config.uuid,
"host_name": None, # PropertyCollector doesn't return host directly
"cluster_name": None,
"datacenter_name": None,
})
except Exception:
continue

container.Destroy()
return vms
