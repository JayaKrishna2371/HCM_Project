class VmwareTemplate(Base):

__tablename__ = "vmware_templates"

id = Column(
Integer,
primary_key=True
)

endpoint_id = Column(
Integer,
nullable=False
)

name = Column(
String,
nullable=False
)

guest_os = Column(
String,
nullable=True
)

cpu_count = Column(
Integer,
nullable=True
)

memory_mb = Column(
Integer,
nullable=True
)


@staticmethod
def save_templates(
endpoint_id: int,
templates: list,
db: Session
):
for t in templates:
db.add(
VmwareTemplate(
endpoint_id=endpoint_id,
name=t.get("name"),
guest_os=t.get("guest_os"),
cpu_count=t.get("cpu"),
memory_mb=t.get("memory_mb")
)
)
db.commit()

@staticmethod
def clear_templates(endpoint_id: int, db: Session):
db.query(VmwareTemplate).filter(
VmwareTemplate.endpoint_id == endpoint_id
).delete()
db.commit()

@staticmethod
def get_templates(endpoint_id: int, db: Session):
return (
db.query(VmwareTemplate)
.filter(VmwareTemplate.endpoint_id == endpoint_id)
.all()
)
